\begin{questions}

\question Given this payoff matrix for the Battle of the Sexes game:

\begin{center}
\begin{tabular}{|c|c|c|}\hline
   & LW     & WL         \\\hline
LW & \textsuperscript{a} 2, 1  & \textsuperscript{b} 0, 0 \\\hline
WL & \textsuperscript{c} 0, 0  & \textsuperscript{d} 1, 2 \\\hline
\end{tabular}
\end{center}

Which cell, or cells is/are a Nash equilibrium?

\begin{solution}[.5in]
a and d
\end{solution}

\question Given this payoff matrix for the Prisoner's Dilemma game:

\begin{center}
\begin{tabular}{|c|c|c|}\hline
  & C      & D         \\\hline
C & \textsuperscript{a} -1, -1 & \textsuperscript{b} -5, 0 \\\hline
D & \textsuperscript{c} 0, -5  & \textsuperscript{d} -3, -3 \\\hline
\end{tabular}
\end{center}

Which cell, or cells is/are a Nash equilibrium?

\begin{solution}[.5in]
d
\end{solution}

\question In Axelrod's famous computer Iterated Prisoner's Dilemma tournaments, which high-level strategy was best, on average?

\begin{solution}[.5in]
Tit For Tat
\end{solution}

\end{questions}
